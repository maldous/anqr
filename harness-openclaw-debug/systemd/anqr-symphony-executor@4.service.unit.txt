# /home/user/.config/systemd/user/anqr-symphony-executor@.service
# anqr-symphony-executor@.service — Worktree-local Symphony executor for worktree %i
#
# Template service for running Symphony executors in isolated worktrees.
# Each instance runs in its own worktree with isolated working directory and git state.
# Performs all AI execution, file mutation, git operations, and validation.
#
# Worktree mapping:
#   wt1-wt4: Reserved for Linear work
#   wt5-wt8: Reserved for Discord work
#
# Port mapping:
#   wt1 → port 4321, wt2 → port 4322, wt3 → port 4323, wt4 → port 4324
#   wt5 → port 4325, wt6 → port 4326, wt7 → port 4327, wt8 → port 4328
#
# Install:
#   cp services/systemd/anqr-symphony-executor@.service ~/.config/systemd/user/
#   systemctl --user daemon-reload
#   systemctl --user enable --now anqr-symphony-executor@{1,2,3,4,5,6,7,8}.service

[Unit]
Description=ANQR Symphony executor for worktree wt%i
After=network-online.target
Wants=network-online.target

[Service]
Type=simple

# CRITICAL: Run in the target worktree directory, not the main repo.
# The script itself is launched from the control-plane checkout so one repo patch
# immediately updates every bridge instance without manually patching each worktree.
WorkingDirectory=%h/src/anqr-wt%i

# Explicit PATH — nvm Node is not on default systemd PATH
Environment=PATH=%h/.nvm/versions/node/v25.8.0/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
Environment=HOME=%h
Environment=NODE_PATH=%h/.nvm/versions/node/v25.8.0/lib/node_modules

# Worktree identification
Environment=ANQR_WORKTREE_ID=wt%i
Environment=ANQR_WORKTREE_PATH=%h/src/anqr-wt%i

# Bridge configuration — port is 4320 + worktree number
Environment=SYMPHONY_BRIDGE_PORT=432%i
# Bridge API key loaded from env file
# Fallback: set SYMPHONY_BRIDGE_API_KEY in anqr.env or anqr-ops.env
Environment=CODEX_SANDBOX_MODE=danger-full-access
Environment=ANQR_RETRIEVAL_WARMUP=false

# Provider tokens (loaded from anqr.env)
EnvironmentFile=-%h/.config/symphony/anqr.env

# Systemd env overrides (LINEAR_API_KEY, etc.)
EnvironmentFile=-%h/.config/systemd/user/anqr-ops.env

ExecStart=%h/.nvm/versions/node/v25.8.0/bin/node --max-old-space-size=2048 --expose-gc %h/src/anqr/harness/ops/integrations/model/symphony-openai-bridge.mjs

# Logging to journal with worktree identifier
StandardOutput=journal
StandardError=journal
SyslogIdentifier=anqr-symphony-executor-wt%i

# Restart on failure with exponential backoff
Restart=on-failure
RestartSec=5

# CRITICAL: Worktree bridges must restart to apply Discord monitoring disable fix
# This prevents duplicate request flooding (4 bridges × 1 request = 4x duplicates)

# Safety limits (same as main bridge)
TimeoutStartSec=30
MemoryHigh=3G
MemoryMax=3584M

[Install]
WantedBy=default.target
